# Local LLM Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Embed Gemma 3 1B on-device for premium users — smart categorization, SMS parsing fallback, monthly summaries, and weekly spending insights.

**Architecture:** `AiService` (singleton, raw inference) is wrapped by `AiProvider` (ChangeNotifier, state machine + isPro gate). Feature adapters in `lib/ai/` are stateless functions that take an `InferFn` parameter for testability. All adapters return `null` on failure so callers fall back silently.

**Tech Stack:** `flutter_gemma` (Google on-device Gemma), Hive (FeedbackStore), SharedPreferences (aiEnabled flag), `path_provider` (already in deps), Provider (already used).

import 'package:hive_flutter/hive_flutter.dart';

class FeedbackStore {
  static const _boxName = 'ai_feedback';

  late Box<String> _box;

  Future<void> init() async {
    _box = await Hive.openBox<String>(_boxName);
  }

  /// Returns the user-corrected category for [merchant], or null if no correction exists.
  String? lookup(String merchant) {
    return _box.get(_key(merchant));
  }

  /// Records that the user chose [userChose] when AI suggested [aiSuggested] for [merchant].
  Future<void> recordCorrection({
    required String merchant,
    required String aiSuggested,
    required String userChose,
  }) async {
    await _box.put(_key(merchant), userChose);
  }

  String _key(String merchant) => merchant.toLowerCase().trim();
}

  static Future<String?> suggest({
    required String merchant,
    required List<String> availableCategories,
    required InferFn infer,
  }) async {
    if (merchant.trim().isEmpty) return null;

    // Deterministic lookup
    final key = merchant.toLowerCase().trim();
    for (final entry in _merchantMap.entries) {
      if (key.contains(entry.key) || entry.key.contains(key)) {
        if (availableCategories.contains(entry.value)) return entry.value;

        '<end_of_turn>\n'
        '<start_of_turn>model\n';

    try {
      final raw = await infer(prompt);
      final result = raw.trim().toLowerCase();
      if (availableCategories.contains(result)) return result;
      return null;
    } catch (_) {
      return null;
    }
  }
}

  SmsParserAdapter._();

  static Future<SmsParseResult?> parse({
    required String rawSms,
    required InferFn infer,
  }) async {
    final prompt = '<start_of_turn>user\n'
        'Extract transaction from this SMS as JSON.\n'
        'SMS: "$rawSms"\n'
        'Reply with only: {"merchant":"...","amount":0.0,"type":"debit|credit"}\n'
        'Unknown fields use null.\n'
        '<end_of_turn>\n'
        '<start_of_turn>model\n';

    try {
      final raw = await infer(prompt);
      final json = jsonDecode(raw.trim()) as Map<String, dynamic>;
      final amountRaw = json['amount'];
      if (amountRaw == null) return null;
      final amount = (amountRaw as num).toDouble();
      if (amount <= 0) return null;
      final type = json['type'] as String? ?? 'debit';
      if (type != 'debit' && type != 'credit') return null;
      return SmsParseResult(
        merchant: json['merchant'] as String?,
        amount: amount,
        type: type,
      );
    } catch (_) {
      return null;
    }
  }
}

**Files:**

`AiProvider` is the state machine. It reads `SubscriptionProvider.isPro` from context, reads/writes `StorageService.aiEnabled`, and drives `AiService`. It exposes `AiModelState`, `downloadProgress`, `isEnabled`, `isReady`.

- [ ] **Step 1: Create AiProvider**

Create `lib/providers/ai_provider.dart`:

```dart
import 'package:flutter/foundation.dart';
import '../services/ai_service.dart';
import '../services/storage_service.dart';

enum AiModelState { notDownloaded, downloading, ready, error }

class AiProvider extends ChangeNotifier {
  final StorageService _storage;

  AiModelState _state = AiModelState.notDownloaded;
  double _downloadProgress = 0.0;
  bool _isPro = false;

  AiModelState get state => _state;
  double get downloadProgress => _downloadProgress;
  bool get isEnabled => _storage.aiEnabled;
  bool get isReady => _state == AiModelState.ready && _isPro;

  AiProvider(this._storage);

  void updateIsPro(bool isPro) {
    if (_isPro == isPro) return;
    _isPro = isPro;
    if (!isPro) {
      // Suspend inference — model file kept
      if (_state == AiModelState.ready) {
        notifyListeners();
      }
    }
  }

  /// Called on app launch. Determines initial state without blocking UI.
  Future<void> init() async {
    if (!_isPro) {
      _state = AiModelState.notDownloaded;
      notifyListeners();
      return;
    }
    if (!isEnabled) {
      _state = AiModelState.notDownloaded;
      notifyListeners();
      return;
    }
    // isEnabled=true → check if model file exists
    final exists = await AiService.instance.modelFileExists;
    if (!exists) {
      // Model was deleted externally — reset enabled flag
      await _storage.setAiEnabled(false);
      _state = AiModelState.notDownloaded;
      notifyListeners();
      return;
    }
    // File exists → init model in background
    _state = AiModelState.ready; // optimistic — set before long init
    notifyListeners();
    try {
      await AiService.instance.initModel();
    } catch (e) {
      debugPrint('[AiProvider] Model init failed: $e');
      _state = AiModelState.error;
      notifyListeners();
    }
  }

  /// Download model then initialize. Called when user taps Download in settings.
  Future<void> enableAndDownload() async {
    if (!_isPro) return;
    _state = AiModelState.downloading;
    _downloadProgress = 0.0;
    notifyListeners();

    try {
      await AiService.instance.downloadModel(
        onProgress: (p) {
          _downloadProgress = p;
          notifyListeners();
        },
      );
      await AiService.instance.initModel();
      await _storage.setAiEnabled(true);
      _state = AiModelState.ready;
    } catch (e) {
      debugPrint('[AiProvider] Download/init failed: $e');
      _state = AiModelState.error;
      await _storage.setAiEnabled(false);
    }
    notifyListeners();
  }

  /// Delete model file and reset state. Called when user taps toggle off.
  Future<void> disable() async {
    await AiService.instance.deleteModel();
    await _storage.setAiEnabled(false);
    _state = AiModelState.notDownloaded;
    _downloadProgress = 0.0;
    notifyListeners();
  }

  /// Reset error state to allow retry.
  void clearError() {
    if (_state == AiModelState.error) {
      _state = AiModelState.notDownloaded;
      notifyListeners();
    }
  }
}
```

```dart
  Future<void> _loadAiContent() async {
    final aiProvider = context.read<AiProvider>();
    final sub = context.read<SubscriptionProvider>();
    if (!aiProvider.isReady || !sub.isPro || _aiContentLoaded) return;
    _aiContentLoaded = true;

    final box = await Hive.openBox('ai_cache');
    final now = DateTime.now();

    // Monthly summary
    final summaryKey = 'ai_summary_${DateFormat('yyyy_MM').format(now)}';
    var summary = box.get(summaryKey) as String?;
    if (summary == null) {
      final txProvider = context.read<TransactionProvider>();
      final month = widget.selectedMonth;
      final monthTx = txProvider.transactions.where((t) =>
          t.date.year == month.year && t.date.month == month.month &&
          t.type == TransactionType.expense).toList();
      final totals = <String, double>{};
      double total = 0;
      for (final t in monthTx) {
        totals[t.categoryId] = (totals[t.categoryId] ?? 0) + t.amount;
        total += t.amount;
      }
      if (totals.isNotEmpty) {
        summary = await SummaryAdapter.generate(
          month: DateFormat('MMMM yyyy').format(month),
          income: 0, // income tracking is future work
          categoryTotals: totals,
          totalSpent: total,
          infer: AiService.instance.infer,
        );
        if (summary != null) await box.put(summaryKey, summary);
      }
    }

    // Weekly insights
    final weekKey = 'ai_insights_${DateFormat('yyyy_ww').format(now)}';
    List<String> insights = [];
    final cachedInsights = box.get(weekKey);
    if (cachedInsights != null) {
      insights = (cachedInsights as List).cast<String>();
    } else {
      final txProvider = context.read<TransactionProvider>();
      final threeMonthsAgo = now.subtract(const Duration(days: 90));
      final recentTx = txProvider.transactions.where((t) =>
          t.date.isAfter(threeMonthsAgo) && t.type == TransactionType.expense).toList();
      final totals = <String, double>{};
      for (final t in recentTx) {
        totals[t.categoryId] = (totals[t.categoryId] ?? 0) + t.amount;
      }
      if (totals.isNotEmpty) {
        for (int i = 0; i < 3; i++) {
          final insight = await InsightAdapter.generate(
            categoryTotals: totals,
            budgetLimits: const {},
            income: 0,
            infer: AiService.instance.infer,
          );
          if (insight != null) insights.add(insight);
        }
        if (insights.isNotEmpty) await box.put(weekKey, insights);
      }
    }

    // Stale insight cleanup (>14 days)
    final staleKeys = box.keys
        .where((k) => k.toString().startsWith('ai_insights_') && k != weekKey)
        .toList();
    for (final k in staleKeys) await box.delete(k);

    if (mounted) setState(() { _aiSummary = summary; _aiInsights = insights; });
  }
```
